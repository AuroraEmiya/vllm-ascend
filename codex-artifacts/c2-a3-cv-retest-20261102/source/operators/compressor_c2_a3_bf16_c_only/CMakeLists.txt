cmake_minimum_required(VERSION 3.20)
project(c2_a3_bf16_c_only LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SOC_VERSION "Ascend910_9382" CACHE STRING "Verified target SoC")
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit")
if(NOT EXISTS "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
    message(FATAL_ERROR "Set ASCEND_HOME_PATH to the verified CANN toolkit")
endif()
include("${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
    OUTPUT_VARIABLE TORCH_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${TORCH_CMAKE}")
find_package(Torch REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import pathlib, torch_npu; print(pathlib.Path(torch_npu.__file__).parent)"
    OUTPUT_VARIABLE TORCH_NPU_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
ascendc_library(c2_a3_bf16_c_only_device STATIC kernel/c_bf16.cpp)
add_library(c2_a3_bf16_c_only SHARED torch_library/register.cpp)
target_compile_options(c2_a3_bf16_c_only PRIVATE ${TORCH_CXX_FLAGS})
target_include_directories(c2_a3_bf16_c_only PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/kernel" "${TORCH_NPU_DIR}/include"
    "${TORCH_NPU_DIR}/include/third_party/acl/inc" "${ASCEND_CANN_PACKAGE_PATH}/include" ${Python3_INCLUDE_DIRS})
target_link_directories(c2_a3_bf16_c_only PRIVATE "${TORCH_NPU_DIR}/lib" "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(c2_a3_bf16_c_only PRIVATE c2_a3_bf16_c_only_device ${TORCH_LIBRARIES} torch_npu ascendcl tiling_api platform)
set_target_properties(c2_a3_bf16_c_only PROPERTIES BUILD_RPATH "${TORCH_NPU_DIR}/lib;${ASCEND_CANN_PACKAGE_PATH}/lib64")
